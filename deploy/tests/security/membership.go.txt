package handler

import (
	"context"
	"net/http/httptest"
	"strings"
	"testing"

	"github.com/go-chi/chi/v5"
	"github.com/tracium/api/internal/middleware"
	"github.com/tracium/api/internal/model"
	"github.com/tracium/api/testing/mocks"
)

type reviewMemberStore struct {
	mocks.MockWorkspaceStore
	grantedUser string
}

func (store *reviewMemberStore) AddMember(ctx context.Context, workspaceID, userID, role string) error {
	store.grantedUser = userID
	return nil
}

type reviewEmailLookup struct{}

func (reviewEmailLookup) ByEmail(ctx context.Context, email string) (*model.User, error) {
	return &model.User{ID: "attacker-account", Email: email}, nil
}

func TestSecurityReviewMembershipTrustsUnverifiedEmail(t *testing.T) {
	store := &reviewMemberStore{MockWorkspaceStore: mocks.MockWorkspaceStore{
		Workspaces: []model.Workspace{mocks.NewTestWorkspace("private-workspace", "owner")},
	}}
	router := chi.NewRouter()
	router.Post("/v1/workspaces/{id}/members", NewWorkspaceHandler(store, reviewEmailLookup{}).AddMember)
	request := httptest.NewRequest("POST", "/v1/workspaces/private-workspace/members", strings.NewReader(`{"email":"colleague@example.test"}`))
	request = request.WithContext(middleware.ContextWithPrincipal(request.Context(), &model.Principal{UserID: "owner"}))
	response := httptest.NewRecorder()
	router.ServeHTTP(response, request)
	if response.Code != 204 || store.grantedUser != "attacker-account" {
		t.Fatalf("previously observed membership grant changed: status=%d grantee=%q", response.Code, store.grantedUser)
	}
	t.Log("CONFIRMED handler behavior: owner supplied a colleague email; lookup account received immediate membership without a mailbox-verification gate (mock store/lookup)")
}
