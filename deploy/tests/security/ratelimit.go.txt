package middleware

import (
	"net/http"
	"net/http/httptest"
	"testing"
	"time"
)

func TestSecurityReviewForwardedHeaderBypass(t *testing.T) {
	limiter := NewRateLimiter(1, time.Minute)
	handler := limiter.Middleware()(http.HandlerFunc(func(writer http.ResponseWriter, request *http.Request) {}))
	statuses := []int{}
	for _, header := range []string{"203.0.113.1, 192.0.2.1", "203.0.113.1, 192.0.2.1", "203.0.113.2, 192.0.2.1", "arbitrary-value, 192.0.2.1"} {
		request := httptest.NewRequest("POST", "/v1/auth/login", nil)
		request.RemoteAddr = "192.0.2.10:12345"
		request.Header.Set("X-Forwarded-For", header)
		response := httptest.NewRecorder()
		handler.ServeHTTP(response, request)
		statuses = append(statuses, response.Code)
	}
	if statuses[0] != 200 || statuses[1] != 429 || statuses[2] != 200 || statuses[3] != 200 {
		t.Fatalf("previously observed bypass changed: %v", statuses)
	}
	t.Logf("CONFIRMED: same socket peer, limit 1; statuses=%v after changing the untrusted forwarded header", statuses)
}
