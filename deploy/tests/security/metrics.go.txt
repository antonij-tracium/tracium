package traciumprocessor

import (
	"context"
	"strings"
	"testing"

	"go.opentelemetry.io/collector/pdata/pmetric"
)

func TestSecurityReviewMetricsBypassSpanBounds(t *testing.T) {
	metrics := pmetric.NewMetrics()
	metricSlice := metrics.ResourceMetrics().AppendEmpty().ScopeMetrics().AppendEmpty().Metrics()
	point := addUsagePoint(metricSlice, "input", "gpt-4o", 1e15)
	point.Attributes().PutStr(attrUserID, strings.Repeat("x", 4096))
	output, err := newMetricsProcessor().processMetrics(context.Background(), metrics)
	if err != nil || output.DataPointCount() != 1 {
		t.Fatalf("previously accepted oversized metric changed: count=%d err=%v", output.DataPointCount(), err)
	}
	cost, _ := point.Attributes().Get(attrCostUSD)
	user, _ := point.Attributes().Get(attrUserID)
	if cost.Double() < 1e9 || len(user.Str()) != 4096 {
		t.Fatalf("previously observed bounds bypass changed: cost=%g user_bytes=%d", cost.Double(), len(user.Str()))
	}
	t.Logf("CONFIRMED: one metric point retained; tokens=1e15 computed_cost_usd=%g user_bytes=%d", cost.Double(), len(user.Str()))
}
