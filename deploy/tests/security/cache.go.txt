package user

import (
	"context"
	"fmt"
	"testing"
)

func TestSecurityReviewUnboundedIdentityCache(t *testing.T) {
	resolver := NewCachedResolver(Passthrough{})
	for index := 0; index < 10000; index++ {
		if _, err := resolver.Resolve(context.Background(), fmt.Sprintf("review-user-%d", index)); err != nil {
			t.Fatal(err)
		}
	}
	retained := 0
	resolver.cache.Range(func(key, value any) bool { retained++; return true })
	if retained != 10000 {
		t.Fatalf("previously observed cache behavior changed: retained=%d", retained)
	}
	t.Logf("CONFIRMED: %d distinct sender identities retained; source has no size limit or eviction", retained)
}
