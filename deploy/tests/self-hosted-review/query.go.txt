package query

import (
	"context"
	"database/sql"
	"database/sql/driver"
	"errors"
	"net/http/httptest"
	"testing"
)

type reviewDeadlineDriver struct{ observed *bool }
type reviewDeadlineConn struct{ observed *bool }

func (probe reviewDeadlineDriver) Open(name string) (driver.Conn, error) {
	return reviewDeadlineConn{observed: probe.observed}, nil
}
func (reviewDeadlineConn) Prepare(query string) (driver.Stmt, error) {
	return nil, errors.New("unused in review")
}
func (reviewDeadlineConn) Close() error { return nil }
func (reviewDeadlineConn) Begin() (driver.Tx, error) {
	return nil, errors.New("unused in review")
}
func (probe reviewDeadlineConn) QueryContext(ctx context.Context, query string, args []driver.NamedValue) (driver.Rows, error) {
	_, deadlineSet := ctx.Deadline()
	*probe.observed = deadlineSet
	return nil, errors.New("review stopped before database execution")
}

func TestSelfHostedReviewNoQueryDeadline(t *testing.T) {
	deadlineSet := true
	sql.Register("self-hosted-deadline-probe", reviewDeadlineDriver{observed: &deadlineSet})
	database, err := sql.Open("self-hosted-deadline-probe", "")
	if err != nil { t.Fatal(err) }
	defer database.Close()
	repository := &ClickHouseRepository{db: database}
	request := httptest.NewRequest("GET", "/v1/traces/review/spans", nil)
	_, _ = repository.GetSpans(request.Context(), "review", []string{"review-workspace"})
	if deadlineSet { t.Fatal("previously observed missing deadline changed") }
	t.Log("CONFIRMED: request context reaches the database driver with no deadline; no network or database execution occurred")
}
