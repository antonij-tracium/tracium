package traciumprocessor

import (
	"context"
	"math"
	"testing"

	"github.com/tracium/collector/internal/pricing"
	"go.opentelemetry.io/collector/pdata/pmetric"
)

func TestReliabilityReviewAggregateTierPricing(t *testing.T) {
	prices, err := pricing.LoadStaticFile("../../../spec/pricing/pricing.json")
	if err != nil { t.Fatal(err) }
	resolver := pricing.NewStaticResolver(prices)
	processor := newMetricsProcessor()
	processor.pricing = resolver
	metrics := pmetric.NewMetrics()
	point := addUsagePoint(metrics.ResourceMetrics().AppendEmpty().ScopeMetrics().AppendEmpty().Metrics(), "input", "gemini-2.5-pro", 300000)
	point.SetCount(2)
	if _, err := processor.processMetrics(context.Background(), metrics); err != nil { t.Fatal(err) }
	perCall, err := resolver.Resolve(context.Background(), "gemini-2.5-pro", pricing.Usage{Input: 150000})
	if err != nil { t.Fatal(err) }
	actual, _ := point.Attributes().Get(attrCostUSD)
	expected := 2 * perCall
	if math.Abs(actual.Double()-expected) < 1e-9 { t.Fatal("previously observed aggregate-tier mismatch changed") }
	t.Logf("CONFIRMED: two 150000-token calls; per-call total=%g USD, aggregated metric cost=%g USD, ratio=%g", expected, actual.Double(), actual.Double()/expected)
}

func TestReliabilityReviewOutputLosesInputTier(t *testing.T) {
	prices, err := pricing.LoadStaticFile("../../../spec/pricing/pricing.json")
	if err != nil { t.Fatal(err) }
	resolver := pricing.NewStaticResolver(prices)
	processor := newMetricsProcessor()
	processor.pricing = resolver
	metrics := pmetric.NewMetrics()
	metricSlice := metrics.ResourceMetrics().AppendEmpty().ScopeMetrics().AppendEmpty().Metrics()
	input := addUsagePoint(metricSlice, "input", "gemini-2.5-pro", 250000)
	output := addUsagePoint(metricSlice, "output", "gemini-2.5-pro", 1000)
	input.SetCount(1)
	output.SetCount(1)
	if _, err := processor.processMetrics(context.Background(), metrics); err != nil { t.Fatal(err) }
	expected, err := resolver.Resolve(context.Background(), "gemini-2.5-pro", pricing.Usage{Input: 250000, Output: 1000})
	if err != nil { t.Fatal(err) }
	inputCost, _ := input.Attributes().Get(attrCostUSD)
	outputCost, _ := output.Attributes().Get(attrCostUSD)
	actual := inputCost.Double() + outputCost.Double()
	if math.Abs(actual-expected) < 1e-9 { t.Fatal("previously observed output-tier mismatch changed") }
	t.Logf("CONFIRMED: one 250000-input/1000-output call; combined usage=%g USD, separate metric points=%g USD", expected, actual)
}
